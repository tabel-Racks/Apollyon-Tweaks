# ---------- CORE TWEAKS ----------

function Set-HighPerformancePlan {
    $highPerf = powercfg -list | Select-String "High performance"
    if ($highPerf) {
        $guid = ($highPerf -split '\s+')[3]
        powercfg -setactive $guid
        [System.Windows.Forms.MessageBox]::Show("Power plan set to High Performance.","Done")
    } else {
        powercfg -duplicatescheme SCHEME_MIN
        [System.Windows.Forms.MessageBox]::Show("High Performance plan duplicated and set.","Done")
    }
}

function Disable-BackgroundApps {
    reg add "HKCU\Software\Microsoft\Windows\CurrentVersion\BackgroundAccessApplications" /v GlobalUserDisabled /t REG_DWORD /d 1 /f | Out-Null
    [System.Windows.Forms.MessageBox]::Show("Background apps disabled.","Done")
}

function Disable-VisualCandy {
    reg add "HKCU\Software\Microsoft\Windows\CurrentVersion\Themes\Personalize" /v EnableTransparency /t REG_DWORD /d 0 /f | Out-Null
    reg add "HKCU\Software\Microsoft\Windows\CurrentVersion\Explorer\VisualEffects" /v VisualFXSetting /t REG_DWORD /d 2 /f | Out-Null
    [System.Windows.Forms.MessageBox]::Show("Visual effects set for performance.","Done")
}

function Disable-Notifications {
    reg add "HKCU\Software\Microsoft\Windows\CurrentVersion\Notifications\Settings" /v NOC_GLOBAL_SETTING_TOASTS_ENABLED /t REG_DWORD /d 0 /f | Out-Null
    [System.Windows.Forms.MessageBox]::Show("Most toast notifications disabled.","Done")
}

function Disable-GameBar {
    reg add "HKCU\Software\Microsoft\GameBar" /v "ShowStartupPanel" /t REG_DWORD /d 0 /f | Out-Null
    reg add "HKCU\Software\Microsoft\GameBar" /v "GameBarEnabled" /t REG_DWORD /d 0 /f | Out-Null
    reg add "HKCU\System\GameConfigStore" /v "GameDVR_FSEBehaviorMode" /t REG_DWORD /d 2 /f | Out-Null
    reg add "HKCU\System\GameConfigStore" /v "GameDVR_HonorUserFSEBehaviorMode" /t REG_DWORD /d 1 /f | Out-Null
    [System.Windows.Forms.MessageBox]::Show("Game Bar & DVR disabled.","Done")
}

function Clean-Temp {
    $paths = @(
        $env:TEMP,
        $env:TMP,
        "$env:SystemRoot\Temp"
    )
    foreach ($p in $paths) {
        if (Test-Path $p) {
            Get-ChildItem $p -Recurse -ErrorAction SilentlyContinue | Remove-Item -Force -Recurse -ErrorAction SilentlyContinue
        }
    }
    [System.Windows.Forms.MessageBox]::Show("Temp folders cleaned.","Done")
}

function Disable-StartupApps {
    $runKey = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run"

    if (Test-Path $runKey) {
        $values = Get-ItemProperty -Path $runKey
        $propertyNames = $values.PSObject.Properties |
            Where-Object { $_.Name -notin @("PSPath","PSParentPath","PSChildName","PSDrive","PSProvider") } |
            Select-Object -ExpandProperty Name

        if ($propertyNames.Count -gt 0) {
            foreach ($name in $propertyNames) {
                Remove-ItemProperty -Path $runKey -Name $name -ErrorAction SilentlyContinue
            }
            [System.Windows.Forms.MessageBox]::Show("User startup apps trimmed.","Done")
        } else {
            [System.Windows.Forms.MessageBox]::Show("No user startup entries found.","Info")
        }
    } else {
        [System.Windows.Forms.MessageBox]::Show("Startup key not found.","Info")
    }
}

# ---------- EXTRA SAFE TWEAKS ----------

function Optimize-NetworkSafe {
    ipconfig /flushdns | Out-Null
    [System.Windows.Forms.MessageBox]::Show("DNS cache flushed. This can help with ping consistency.","Network")
}

function Clean-GameCachesSafe {
    $paths = @()

    # DirectX shader cache
    $dxCache = "$env:LOCALAPPDATA\D3DSCache"
    if (Test-Path $dxCache) { $paths += $dxCache }

    # Epic / Fortnite common cache locations (safe check only)
    $fortCache1 = "$env:LOCALAPPDATA\FortniteGame\Saved\Logs"
    $fortCache2 = "$env:LOCALAPPDATA\FortniteGame\Saved\Temp"
    if (Test-Path $fortCache1) { $paths += $fortCache1 }
    if (Test-Path $fortCache2) { $paths += $fortCache2 }

    foreach ($p in $paths) {
        if (Test-Path $p) {
            Get-ChildItem $p -Recurse -ErrorAction SilentlyContinue | Remove-Item -Force -Recurse -ErrorAction SilentlyContinue
        }
    }

    [System.Windows.Forms.MessageBox]::Show("Shader & game temp caches cleaned (where found).","Game Cache")
}
